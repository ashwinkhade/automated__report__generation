## DDL (auto-generated by SQLAlchemy)

```sql
CREATE TABLE users (
  id            SERIAL PRIMARY KEY,
  email         VARCHAR(255) UNIQUE NOT NULL,
  username      VARCHAR(100) UNIQUE NOT NULL,
  hashed_password VARCHAR(255) NOT NULL,
  full_name     VARCHAR(200),
  is_active     BOOLEAN DEFAULT TRUE,
  is_admin      BOOLEAN DEFAULT FALSE,
  created_at    TIMESTAMPTZ DEFAULT now(),
  updated_at    TIMESTAMPTZ
);

CREATE TABLE datasets (
  id            SERIAL PRIMARY KEY,
  owner_id      INTEGER NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  name          VARCHAR(255) NOT NULL,
  file_path     VARCHAR(500) NOT NULL,
  s3_key        VARCHAR(500),
  file_type     VARCHAR(20),
  size_bytes    BIGINT DEFAULT 0,
  row_count     INTEGER DEFAULT 0,
  column_count  INTEGER DEFAULT 0,
  columns_meta  JSONB DEFAULT '[]',
  created_at    TIMESTAMPTZ DEFAULT now()
);

CREATE TABLE reports (
  id              SERIAL PRIMARY KEY,
  owner_id        INTEGER NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  dataset_id      INTEGER REFERENCES datasets(id) ON DELETE CASCADE,
  title           VARCHAR(255) NOT NULL,
  status          VARCHAR(20) DEFAULT 'pending',
  summary         TEXT,
  insights        JSONB DEFAULT '[]',
  recommendations JSONB DEFAULT '[]',
  kpis            JSONB DEFAULT '{}',
  charts          JSONB DEFAULT '[]',
  raw_analytics   JSONB DEFAULT '{}',
  html_path       VARCHAR(500),
  pdf_path        VARCHAR(500),
  docx_path       VARCHAR(500),
  error_message   TEXT,
  created_at      TIMESTAMPTZ DEFAULT now(),
  completed_at    TIMESTAMPTZ
);

CREATE INDEX idx_datasets_owner ON datasets(owner_id);
CREATE INDEX idx_reports_owner ON reports(owner_id);
CREATE INDEX idx_reports_status ON reports(status);
```
